#+title: Інструмент items

Робота з елементами сейфа (~ciphers~) через ~VaultClient::ciphers()~ (див. [[file:main-client.org][~main-client.org~]]): ~list~
(перелік), ~get~ (повний розшифрований обʼєкт), ~create~ (новий елемент — ~login~ чи ~secureNote~) і
~edit~ (записати зміну, зберігши решту полів як є). ~card~ / ~identity~ поки не підтримано — додамо,
коли з'явиться реальна потреба. Усі чотири дії залежать від власного форку ~sdk-internal~ (див.
[[file:config.org][~config.org~]]) — на апстрімі жодна б не спрацювала: немає ні синхронізації ~ciphers~, ні
реекспорту типів запиту для ~create~ / ~edit~.

Перевірено на реальному перенесенні 271 картки (~it-manage~, 2026-09-24) — 271/271, 0 розбіжностей
на незалежній звірці. Виявлена межа: елементи з непорожнім ~organizationId~ (спільні org-колекції)
падають на ~edit~ з ~"Missing Key for Id: Organization(<id>)"~ — ~CipherSyncHandler~ синхронізує
особисті ~ciphers~, але не тримає/не розшифровує org-ключ, потрібний для org-власних елементів. Не
блокер (такі елементи й далі йдуть через ~bw~ CLI), але наступна прогалина форку після базового
~items~, якщо колись знадобиться повна паритетність.

* Приховування секретів

Жодна дія, крім ~rotate_secret~ (нижче), не повертає ~login.password~/~login.totp~ агенту — навіть
коли сама дія їх не змінювала (напр. ~get~/~edit~ повертають повний обʼєкт із уже наявним паролем).
Так само ніколи не повертається ~sshKey.privateKey~ (елемент типу ~5~, =ssh key=): публічний ключ
і відбиток лишаються. Прибираємо ці поля рекурсивно з уже серіалізованого ~JSON~, а не з самого типу
— так самий фільтр працює і для повного ~CipherView~ (~get~/~create~/~edit~), і для переліку
~CipherListView~, чия форма енкодування варіанта ~CipherListViewType~ наперед невідома.

#+begin_src rust :noweb-ref Визначення
  fn redact(value: &mut serde_json::Value) {
      match value {
          serde_json::Value::Object(fields) => {
              fields.remove("password");
              fields.remove("totp");
              fields.remove("privateKey");

              for field in fields.values_mut() {
                  redact(field);
              }
          }
          serde_json::Value::Array(items) => {
              for item in items {
                  redact(item);
              }
          }
          _ => {}
      }
  }
#+end_src

* Перелік

Відповідь — обʼєкт ~{ items, failures }~: ~items~ — успішно розшифровані елементи (~successes~), ~failures~
— лише ~id~, ~type~ і ~folderId~ елементів, які SDK не зміг розшифрувати чи розібрати (вміст зашифрований і
не повертається). Раніше такі елементи мовчки відкидались — так зникали елементи типу ~5~ (=ssh key=).
Фільтри ~name~/~folderId~ застосовуються лише до ~items~. ~CipherListView~ і так
не має поля ~password~ (перевірено рядком-у-рядок з ~cargo doc~, див.
~spcfctn/code/bitwarden-rust-sdk.md~), але ~totp~ там є — тож ~redact~ однаково потрібен.

#+begin_src rust :noweb-ref Визначення
  async fn list(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let result = bw.vault().ciphers().list().await.map_err(|error| error.to_string())?;
      let name = arguments["name"].as_str().map(|name| name.to_lowercase());
      let folder = arguments["folderId"].as_str();

      let mut items = serde_json::to_value(&result.successes).unwrap();
      items.as_array_mut().unwrap().retain(|item| {
          name.as_ref().map_or(true, |name| item["name"].as_str().unwrap().to_lowercase().contains(name))
              && folder.map_or(true, |folder| item["folderId"].as_str() == Some(folder))
      });
      redact(&mut items);

      let failures: Vec<serde_json::Value> = result
          .failures
          .iter()
          .map(|cipher| {
              let value = serde_json::to_value(cipher).unwrap();
              serde_json::json!({ "id": value["id"], "type": value["type"], "folderId": value["folderId"] })
          })
          .collect();

      Ok(serde_json::json!({ "items": items, "failures": failures }).to_string())
  }
#+end_src

Необовʼязкові фільтри ~name~ (підрядок без урахування регістру) і ~folderId~ (точний збіг) звужують
перелік ще до серіалізації відповіді: повний перелік сейфа — понад мегабайт, і агенту він зазвичай не
потрібен.

* Синхронізація

~bwm-mcp~ синхронізується з сервером один раз при старті процесу (~client::connect~), тож елементи,
створені чи змінені деінде (клієнтом ~Bitwarden~, іншою сесією) після цього не видно, доки процес не
перезапущено. Дія ~sync~ виконує повторну синхронізацію (~force: true~) без перепідключення й повторного
входу.

#+begin_src rust :noweb-ref Визначення
  async fn resync(bw: &client::Bw) -> Result<String, String> {
      client::sync(bw, true).await?;

      Ok("Синхронізовано.".to_string())
  }
#+end_src

* Створення

Підтримано три типи — ~login~ (~type: 1~), ~secureNote~ (~type: 2~) і =ssh key= (~type: 5~) — той самий
набір значень, що й у офіційному сервері. ~card~ / ~identity~ не підтримано: аргумент ~type~ поза
~{1, 2, 5}~ дає зрозумілу помилку, а не незрозумілий збій десь глибше в ~CipherViewType~.

Ключ =ssh= (~type: 5~) завжди генерується самим ~bwm-mcp~ (~bitwarden_ssh::generator::generate_sshkey~,
алгоритм ~sshKey.algorithm~, типово ~ed25519~) і одразу йде в елемент: приватна частина не передається
агентом і не повертається у відповіді (~redact~), публічний ключ і відбиток — повертаються.

Агент не може передати готове значення ~login.password~ (і взагалі не може ~login.totp~ — генератора
~totp~-секрету в =SDK= немає, на відміну від пароля): єдиний спосіб отримати пароль при створенні —
попросити ~bwm-mcp~ згенерувати його самому (~login.generatePassword~), і сам згенерований пароль
назад не повертається (той самий ~redact~, що й для ~get~).

  fn uri_match(value: &serde_json::Value) -> Option<UriMatchType> {
      match value.as_i64() {
          Some(0) => Some(UriMatchType::Domain),
          Some(1) => Some(UriMatchType::Host),
          Some(2) => Some(UriMatchType::StartsWith),
          Some(3) => Some(UriMatchType::Exact),
          Some(4) => Some(UriMatchType::RegularExpression),
          Some(5) => Some(UriMatchType::Never),
          _ => None,
      }
  }

  fn uris(arguments: &serde_json::Value) -> Option<Vec<LoginUriView>> {
      let array = arguments["login"]["uris"].as_array()?;

      Some(
          array
              .iter()
              .map(|item| LoginUriView {
                  uri: item["uri"].as_str().map(|uri| uri.to_string()),
                  r#match: uri_match(&item["match"]),
                  uri_checksum: None,
              })

      )

  fn login_view(arguments: &serde_json::Value) -> LoginView {
      LoginView {
          username: arguments["login"]["username"].as_str().map(|username| username.to_string()),
          password: None,
          password_revision_date: None,
          uris: uris(arguments),
          totp: None,
          autofill_on_page_load: None,
          fido2_credentials: None,
      }

  fn generate_key(algorithm: Option<&str>) -> Result<SshKeyView, String> {
      let algorithm = match algorithm.unwrap_or("ed25519") {
          "ed25519" => KeyAlgorithm::Ed25519,
          "rsa3072" => KeyAlgorithm::Rsa3072,
          "rsa4096" => KeyAlgorithm::Rsa4096,
          "ecdsap256" => KeyAlgorithm::EcdsaP256,
          "ecdsap384" => KeyAlgorithm::EcdsaP384,
          "ecdsap521" => KeyAlgorithm::EcdsaP521,
          _ => return Err("sshKey.algorithm: ed25519/rsa3072/rsa4096/ecdsap256/ecdsap384/ecdsap521.".to_string()),
      };
      let key = generate_sshkey(algorithm).map_err(|error| error.to_string())?;

      Ok(SshKeyView { private_key: key.private_key, public_key: key.public_key, fingerprint: key.fingerprint })
  }

  fn cipher_type(arguments: &serde_json::Value) -> Result<CipherViewType, String> {
      match arguments["type"].as_i64() {
          Some(1) => Ok(CipherViewType::Login(login_view(arguments))),
          Some(2) => Ok(CipherViewType::SecureNote(SecureNoteView { r#type: SecureNoteType::Generic })),
          Some(5) => Ok(CipherViewType::SshKey(generate_key(arguments["sshKey"]["algorithm"].as_str())?)),
          _ => Err("Потрібен аргумент type: 1 (login), 2 (secureNote) або 5 (sshKey).".to_string()),
      }
  }

  fn generate_password(
      bw: &client::Bw,
      arguments: &serde_json::Value,
      cipher_type: &mut CipherViewType,
  ) -> Result<(), String> {
      if !arguments["login"]["generatePassword"].as_bool().unwrap_or(false) {
          return Ok(());
      }

      let CipherViewType::Login(login) = cipher_type else {
          return Err("login.generatePassword можливий лише для type=1 (login).".to_string());
      };

      let request = PasswordGeneratorRequest {
          lowercase: true,
          uppercase: true,
          numbers: true,
          special: false,
          length: 20,
          ..Default::default()
      };

      login.password = Some(bw.generator().password(request).map_err(|error| error.to_string())?);

      Ok(())
  }

  async fn create(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let name = arguments["name"].as_str().ok_or("Потрібен аргумент name.".to_string())?.to_string();
      let mut r#type = cipher_type(arguments)?;
      generate_password(bw, arguments, &mut r#type)?;
      let folder_id = match arguments["folderId"].as_str() {
          Some(id) => Some(id.parse().map_err(|_| "Некоректний folderId.".to_string())?),
          None => None,
      };

      let request = CipherCreateRequest {
          organization_id: None,
          collection_ids: vec![],
          folder_id,
          name,
          notes: arguments["notes"].as_str().map(|notes| notes.to_string()),
          favorite: false,
          reprompt: CipherRepromptType::default(),
          r#type,
          fields: vec![],
          archived_date: None,
      };

      let created = bw.vault().ciphers().create(request).await.map_err(|error| error.to_string())?;
      let mut value = serde_json::to_value(&created).unwrap();
      redact(&mut value);

      Ok(value.to_string())
  }

* Редагування

Спершу отримуємо поточний повний обʼєкт і перетворюємо на ~CipherEditRequest~
(~TryFrom<CipherView>~) — так усі поля (~login~, ~fields~, ~attachments~ тощо) лишаються як є, крім
тих, що прийшли в аргументах явно. Поки підтримано лише ~folderId~ / ~name~ / ~notes~ — ~login~ /
~card~ / ~identity~ додамо, коли з'явиться реальна потреба редагувати саме їх.

  async fn edit(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let id = item_id(arguments)?;
      let view = bw.vault().ciphers().get(&id).await.map_err(|error| error.to_string())?;
      let mut request = CipherEditRequest::try_from(view).map_err(|error| error.to_string())?;

      if let Some(folder_id) = arguments["folderId"].as_str() {
          request.folder_id = Some(folder_id.parse().map_err(|_| "Некоректний folderId.".to_string())?);
      }
      if let Some(name) = arguments["name"].as_str() {
          request.name = name.to_string();
      }
      if let Some(notes) = arguments["notes"].as_str() {
          request.notes = Some(notes.to_string());
      }

      let updated = bw.vault().ciphers().edit(request).await.map_err(|error| error.to_string())?;
      let mut value = serde_json::to_value(&updated).unwrap();
      redact(&mut value);

      Ok(value.to_string())
  }

* Ротація секрету

Єдина дія, що встановлює нове значення ~login.password~/~login.totp~ — навмисно окрема від
звичайного ~edit~ (не додає ~login~ у його аргументи), щоб пізніше її можна було обмежити окремим
правилом (без ризику зачепити звичайне редагування ~folderId~/~name~/~notes~). На відміну від решти
дій, відповідь для ~login~ тут ~redact~ не проходить — покликач сам щойно передав ці значення,
приховувати нічого. Для елемента =ssh key= (~type: 5~) ротація — повторна генерація ключа всередині
~bwm-mcp~ (~secret.algorithm~, типово ~ed25519~): агент нового ключа не бачив, тож відповідь ~redact~
проходить, приватної частини в ній нема.

#+begin_src rust :noweb-ref Визначення
  async fn rotate_secret(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let id = item_id(arguments)?;
      let view = bw.vault().ciphers().get(&id).await.map_err(|error| error.to_string())?;
      let mut request = CipherEditRequest::try_from(view).map_err(|error| error.to_string())?;

      if let CipherViewType::SshKey(key) = &mut request.r#type {
          *key = generate_key(arguments["secret"]["algorithm"].as_str())?;
          let updated = bw.vault().ciphers().edit(request).await.map_err(|error| error.to_string())?;
          let mut value = serde_json::to_value(&updated).unwrap();
          redact(&mut value);

          return Ok(value.to_string());
      }

      let CipherViewType::Login(login) = &mut request.r#type else {
          return Err("rotate_secret можливий лише для елементів login і sshKey.".to_string());
      };

      let password = arguments["secret"]["password"].as_str();
      let totp = arguments["secret"]["totp"].as_str();

      if password.is_none() && totp.is_none() {
          return Err("Потрібен хоча б secret.password чи secret.totp.".to_string());
      }
      if let Some(password) = password {
          login.password = Some(password.to_string());
      }
      if let Some(totp) = totp {
          login.totp = Some(totp.to_string());
      }

      let updated = bw.vault().ciphers().edit(request).await.map_err(|error| error.to_string())?;

      Ok(serde_json::to_string(&updated).unwrap())
  }

  pub async fn run(bw: &client::Bw, arguments: &serde_json::Value) -> serde_json::Value {
      let result = match arguments["action"].as_str() {
          Some("list") => list(bw, arguments).await,
          Some("sync") => resync(bw).await,
          Some("get") => get(bw, arguments).await,
          Some("create") => create(bw, arguments).await,

          _ => Err("Потрібен аргумент action: list/get/create/edit/rotate_secret/sync.".to_string()),
      };

  }
