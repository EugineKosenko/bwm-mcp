#+title: Інструмент items

Робота з елементами сейфа (~ciphers~) через ~VaultClient::ciphers()~ (див. [[file:main-client.org][~main-client.org~]]): ~list~
(перелік), ~get~ (повний розшифрований обʼєкт), ~create~ (новий елемент — ~login~ чи ~secureNote~) і
~edit~ (записати зміну, зберігши решту полів як є). ~card~ / ~identity~ поки не підтримано — додамо, коли
з'явиться реальна потреба. Усі чотири дії залежать від власного форку ~sdk-internal~ (див.  [[file:config.org][~config.org~]])
— на апстрімі жодна б не спрацювала: немає ні синхронізації ~ciphers~, ні реекспорту типів запиту для
~create~ / ~edit~.

Перевірено на реальному перенесенні 271 картки (~it-manage~, 2026-09-24) — 271/271, 0 розбіжностей на
незалежній звірці. Виявлена межа: елементи з непорожнім ~organizationId~ (спільні org-колекції) падають
на ~edit~ з ~"Missing Key for Id: Organization(<id>)"~ — ~CipherSyncHandler~ синхронізує особисті
~ciphers~, але не тримає/не розшифровує org-ключ, потрібний для org-власних елементів. Не блокер (такі
елементи й далі йдуть через ~bw~ CLI), але наступна прогалина форку після базового ~items~, якщо колись
знадобиться повна паритетність.

* Приховування секретів

Жодна дія, крім ~rotate_secret~ (нижче), не повертає ~login.password~/~login.totp~ агенту — навіть коли
сама дія їх не змінювала (напр. ~get~/~edit~ повертають повний обʼєкт із уже наявним паролем).  Так само
ніколи не повертається ~sshKey.privateKey~ (елемент типу ~5~, =ssh key=): публічний ключ і відбиток
лишаються. Прибираємо ці поля рекурсивно з уже серіалізованого ~JSON~, а не з самого типу — так самий
фільтр працює і для повного ~CipherView~ (~get~/~create~/~edit~), і для переліку ~CipherListView~, чия
форма енкодування варіанта ~CipherListViewType~ наперед невідома.

#+begin_src rust :noweb-ref Визначення
  fn redact(value: &mut serde_json::Value) {
      match value {
          serde_json::Value::Object(fields) => {
              fields.remove("password");
              fields.remove("totp");
              fields.remove("privateKey");

              for field in fields.values_mut() {
                  redact(field);
              }
          }
          serde_json::Value::Array(items) => {
              for item in items {
                  redact(item);
              }
          }
          _ => {}
      }
  }
#+end_src

* Логін підключення (тип sshKey)

Для елемента =ssh key= (~type: 5~) поле ~SshKeyView~ не має місця для рядка "як підключатись" — лише
~private_key~/~public_key~/~fingerprint~ (перевірено ~cargo doc~,
~spcfctn/code/bitwarden-rust-sdk.md~). Тому логін для підключення (~<користувач>@<IPv4>~, напр.
~azureuser@52.188.198.115~ — лише ~IPv4~, без ~FQDN~/~IPv6~/кількох адрес, за домовленістю з ~it-manage~)
зберігається як звичайне користувацьке поле ~fields[]~ з іменем ~"login"~, але структуровано (=bwm-mcp=
сам його пише/читає/валідує), а не як довільне поле, віддане на розсуд
агента. ~get~/~create~/~edit~/~rotate_secret~ виносять це поле окремим ключем верхнього рівня ~login~ у
відповіді — не всередині загального ~fields~. Для ~list~ це структурно неможливо: ~CipherListView~ (на
відміну від повного ~CipherView~) взагалі не має поля ~fields~ — тож ~list~ логін не показує; пошук на
кшталт "усі входи на 127.0.0.1" через ~list~ не працює, споживач мусить сам дістати й відфільтрувати
елементи через ~get~.

#+begin_src rust :noweb-ref Визначення
  fn ssh_login(value: &str) -> Result<String, String> {
      let (_user, host) = value
          .split_once('@')
          .filter(|(user, _)| !user.is_empty())
          .ok_or("sshKey.login: очікується формат <користувач>@<IPv4>.".to_string())?;

      let octets: Vec<&str> = host.split('.').collect();
      let ipv4 = octets.len() == 4 && octets.iter().all(|octet| octet.parse::<u8>().is_ok());

      if !ipv4 {
          return Err("sshKey.login: <IPv4> — чотири числа 0-255 через крапку, без FQDN/IPv6.".to_string());
      }

      Ok(value.to_string())
  }

  fn login_field(login: String) -> FieldView {
      FieldView { name: Some("login".to_string()), value: Some(login), r#type: FieldType::Text, linked_id: None }
  }

  fn promote_login(value: &mut serde_json::Value) {
      let Some(fields) = value.get_mut("fields").and_then(|fields| fields.as_array_mut()) else { return };

      if let Some(index) = fields.iter().position(|field| field["name"].as_str() == Some("login")) {
          let field = fields.remove(index);
          value["login"] = field["value"].clone();
      }
  }

* Перелік

Відповідь — обʼєкт ~{ items, failures }~: ~items~ — успішно розшифровані елементи (~successes~),
~failures~ — лише ~id~, ~type~ і ~folderId~ елементів, які SDK не зміг розшифрувати чи розібрати (вміст
зашифрований і не повертається). Раніше такі елементи мовчки відкидались — так зникали елементи типу ~5~
(=ssh key=).  Фільтри ~name~/~folderId~ застосовуються лише до ~items~. ~CipherListView~ і так не має
поля ~password~ (перевірено рядком-у-рядок з ~cargo doc~, див.  ~spcfctn/code/bitwarden-rust-sdk.md~),
але ~totp~ там є — тож ~redact~ однаково потрібен.

#+begin_src rust :noweb-ref Визначення
  async fn list(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let result = bw.vault().ciphers().list().await.map_err(|error| error.to_string())?;
      let name = arguments["name"].as_str().map(|name| name.to_lowercase());
      let folder = arguments["folderId"].as_str();

      let mut items = serde_json::to_value(&result.successes).unwrap();
      items.as_array_mut().unwrap().retain(|item| {
          name.as_ref().map_or(true, |name| item["name"].as_str().unwrap().to_lowercase().contains(name))
              && folder.map_or(true, |folder| item["folderId"].as_str() == Some(folder))
      });
      redact(&mut items);

      let failures: Vec<serde_json::Value> = result
          .failures
          .iter()
          .map(|cipher| {
              let value = serde_json::to_value(cipher).unwrap();
              serde_json::json!({ "id": value["id"], "type": value["type"], "folderId": value["folderId"] })
          })
          .collect();

      Ok(serde_json::json!({ "items": items, "failures": failures }).to_string())
  }

* Створення

Підтримано три типи — ~login~ (~type: 1~), ~secureNote~ (~type: 2~) і =ssh key= (~type: 5~) — той самий
набір значень, що й у офіційному сервері. ~card~ / ~identity~ не підтримано: аргумент ~type~ поза ~{1, 2,
5}~ дає зрозумілу помилку, а не незрозумілий збій десь глибше в ~CipherViewType~.

Ключ =ssh= (~type: 5~) завжди генерується самим ~bwm-mcp~ (~bitwarden_ssh::generator::generate_sshkey~,
алгоритм ~sshKey.algorithm~, типово ~ed25519~) і одразу йде в елемент: приватна частина не передається
агентом і не повертається у відповіді (~redact~), публічний ключ і відбиток — повертаються.

Агент не може передати готове значення ~login.password~ (і взагалі не може ~login.totp~ — генератора
~totp~-секрету в =SDK= немає, на відміну від пароля): єдиний спосіб отримати пароль при створенні —
попросити ~bwm-mcp~ згенерувати його самому (~login.generatePassword~), і сам згенерований пароль назад
не повертається (той самий ~redact~, що й для ~get~).

Для =ssh key= (~type: 5~) додатково обов'язковий ~sshKey.login~ (~<користувач>@<IPv4>~, див. розділ вище)
— без нього ~create~ відмовляє, а не створює елемент без прив'язки до конкретного хоста.

  fn generate_password(
      bw: &client::Bw,
      arguments: &serde_json::Value,
      cipher_type: &mut CipherViewType,
  ) -> Result<(), String> {
      if !arguments["login"]["generatePassword"].as_bool().unwrap_or(false) {
          return Ok(());
      }

      let CipherViewType::Login(login) = cipher_type else {
          return Err("login.generatePassword можливий лише для type=1 (login).".to_string());
      };

      let request = PasswordGeneratorRequest {
          lowercase: true,
          uppercase: true,
          numbers: true,
          special: false,
          length: 20,
          ..Default::default()
      };

      login.password = Some(bw.generator().password(request).map_err(|error| error.to_string())?);

      Ok(())
  }

* Редагування

Спершу отримуємо поточний повний обʼєкт і перетворюємо на ~CipherEditRequest~ (~TryFrom<CipherView>~) —
так усі поля (~login~, ~fields~, ~attachments~ тощо) лишаються як є, крім тих, що прийшли в аргументах
явно. З типового ~login~/~card~/~identity~ поки підтримано лише ~folderId~/~name~/~notes~ — додамо решту,
коли з'явиться реальна потреба. Виняток — ~sshKey.login~ (розділ вище): опціональний, але коли переданий,
справді замінює збережене значення (валідація тим самим форматом, що й при ~create~).

* Ротація секрету

Єдина дія, що встановлює нове значення ~login.password~/~login.totp~ — навмисно окрема від звичайного
~edit~ (не додає ~login~ у його аргументи), щоб пізніше її можна було обмежити окремим правилом (без
ризику зачепити звичайне редагування ~folderId~/~name~/~notes~). Для елемента =ssh key= (~type: 5~)
ротація — повторна генерація ключа всередині ~bwm-mcp~ (~secret.algorithm~, типово ~ed25519~): агент
нового ключа не бачив, тож відповідь ~redact~ проходить, приватної частини в ній нема.

Для ~login~ — два незалежні способи задати новий пароль: ~secret.password~ (готове значення, покликач
сам щойно передав його — відповідь для ~password~ тоді НЕ приховується, приховувати нічого) і
~secret.generatePassword: true~ (генерація на боці сервера з тими самими опціями, що в інструменті
~generate~ — ~length~/~lowercase~/~uppercase~/~number~/~special~, нижче в
[[file:main-tools.org][~main-tools.org~]]) — агент тоді значення не бачив узагалі, тож саме ~password~ приховується в
відповіді, хоч решта поля лишається (ціль — замінити секрет "на місці", без витоку нового значення в
контекст агента і без створення другої картки). Обидва разом — помилка. ~secret.totp~ лишається лише
явним значенням: генератора =totp=-секрету в =SDK= немає (див. [[file:config.org][~config.org~]]), тож
"~generateTotp~" поки неможливий.

  async fn rotate_secret(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let id = item_id(arguments)?;
      let view = bw.vault().ciphers().get(&id).await.map_err(|error| error.to_string())?;
      let mut request = CipherEditRequest::try_from(view).map_err(|error| error.to_string())?;

      if let CipherViewType::SshKey(key) = &mut request.r#type {
          ,*key = generate_key(arguments["secret"]["algorithm"].as_str())?;
          let updated = bw.vault().ciphers().edit(request).await.map_err(|error| error.to_string())?;
          let mut value = serde_json::to_value(&updated).unwrap();
          promote_login(&mut value);
          redact(&mut value);

          return Ok(value.to_string());
      }

      let CipherViewType::Login(login) = &mut request.r#type else {
          return Err("rotate_secret можливий лише для елементів login і sshKey.".to_string());
      };

      let password = arguments["secret"]["password"].as_str();
      let generate = arguments["secret"]["generatePassword"].as_bool().unwrap_or(false);
      let totp = arguments["secret"]["totp"].as_str();

      if password.is_some() && generate {
          return Err("secret.password і secret.generatePassword взаємовиключні.".to_string());
      }
      if password.is_none() && !generate && totp.is_none() {
          return Err("Потрібен хоча б secret.password, secret.generatePassword чи secret.totp.".to_string());
      }
      if let Some(password) = password {
          login.password = Some(password.to_string());
      }
      if generate {
          let request = password_request(&arguments["secret"]);
          login.password = Some(bw.generator().password(request).map_err(|error| error.to_string())?);
      }
      if let Some(totp) = totp {
          login.totp = Some(totp.to_string());
      }

      let updated = bw.vault().ciphers().edit(request).await.map_err(|error| error.to_string())?;
      let mut value = serde_json::to_value(&updated).unwrap();
      promote_login(&mut value);

      if generate {
          redact_password(&mut value);
      }

      Ok(value.to_string())
  }
#+end_src

* Кошик

~soft_delete~/~restore~ (крейт ~bitwarden-vault~, перевірено ~cargo doc~) — м'яке видалення: елемент іде в
кошик, не зникає назавжди, і його можна повернути. Повне (``delete``/``delete_many``) свідомо не
підключено — немає підтвердженого сценарію, що потребує саме незворотного видалення, а ризик вищий.

#+begin_src rust :noweb-ref Декларації

#+begin_src rust :noweb-ref Визначення
  fn cipher_id(id: &str) -> Result<CipherId, String> {
      id.parse().map_err(|_| "Некоректний id.".to_string())
  }

  async fn delete(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let id = cipher_id(&item_id(arguments)?)?;
      bw.vault().ciphers().soft_delete(id).await.map_err(|error| error.to_string())?;

      Ok("Перенесено в кошик.".to_string())
  }

  async fn restore(bw: &client::Bw, arguments: &serde_json::Value) -> Result<String, String> {
      let id = cipher_id(&item_id(arguments)?)?;
      let restored = bw.vault().ciphers().restore(id).await.map_err(|error| error.to_string())?;
      let mut value = serde_json::to_value(&restored).unwrap();
      promote_login(&mut value);
      redact(&mut value);

      Ok(value.to_string())
  }
